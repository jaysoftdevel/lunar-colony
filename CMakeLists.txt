cmake_minimum_required(VERSION 3.20)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" GAME_VERSION_FULL LIMIT_COUNT 1)
if(NOT GAME_VERSION_FULL MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must contain release.major.minor, e.g. 0.19.2")
endif()
project(game_workspace VERSION ${GAME_VERSION_FULL} LANGUAGES CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED CMAKE_ARCHIVE_OUTPUT_DIRECTORY)
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
endif()
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Game build configuration" FORCE)
endif()
option(GAME_BUILD_NAVIGATION_TESTS "Build headless navigation regressions" OFF)

if(WIN32)
    set(GAME_PLATFORM_NAME "windows")
    set(GAME_RENDER_BACKEND "d3d12")
elseif(UNIX AND NOT APPLE)
    set(GAME_PLATFORM_NAME "linux")
    set(GAME_RENDER_BACKEND "opengl")
else()
    message(FATAL_ERROR "Unsupported host platform. Supported platforms: Windows 11 and Linux/X11.")
endif()
message(STATUS "Game platform: ${GAME_PLATFORM_NAME}; renderer: ${GAME_RENDER_BACKEND}")
if(GAME_BUILD_NAVIGATION_TESTS)
    enable_testing()
endif()
add_subdirectory(engine)
add_subdirectory(third_party/rvo2)
add_subdirectory(game)
